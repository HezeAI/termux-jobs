cmake_minimum_required(VERSION 3.20)
project(JobsLocalAi LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH TRUE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
if(NOT JOBS_LLAMA_SOURCE)
    message(FATAL_ERROR "Set JOBS_LLAMA_SOURCE to the prepared llama.cpp-b9222 directory")
endif()
option(JOBS_FAST_CPU "Build guarded ARM dotprod/fp16/i8mm variant" OFF)
if(JOBS_FAST_CPU)
 set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16+i8mm" CACHE STRING "" FORCE)
 set(JOBS_TARGET jobs_llama_fast)
else()
 set(GGML_CPU_ARM_ARCH "" CACHE STRING "" FORCE)
 set(JOBS_TARGET jobs_llama)
endif()
add_subdirectory("${JOBS_LLAMA_SOURCE}" llama)
add_library(${JOBS_TARGET} SHARED jobs-llama.cpp)
if(JOBS_FAST_CPU)
 target_compile_definitions(${JOBS_TARGET} PRIVATE JOBS_FAST_CPU=1)
endif()
target_link_libraries(${JOBS_TARGET} PRIVATE llama)
target_link_options(${JOBS_TARGET} PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/jobs-exports.map" -Wl,-z,max-page-size=16384 -Wl,--no-undefined)

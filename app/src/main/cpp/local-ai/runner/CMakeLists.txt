cmake_minimum_required(VERSION 3.20)
project(JobsLlamaRunner LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH TRUE)
foreach(flag BUILD_SHARED_LIBS GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
if(JOBS_FAST_CPU)
 set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16+i8mm" CACHE STRING "" FORCE)
else()
 set(GGML_CPU_ARM_ARCH "" CACHE STRING "" FORCE)
endif()
add_subdirectory("${JOBS_LLAMA_SOURCE}" llama)
add_executable(jobs_runner jobs-runner.cpp)
target_include_directories(jobs_runner PRIVATE .. "${JOBS_LLAMA_SOURCE}/vendor")
target_link_libraries(jobs_runner PRIVATE llama)
